struct {
    int val;
    TreeNode* left;

2.二叉树遍历

  • 144.前序遍历（中左右）​

递归​

  1. 递归传入参数为：当前节点的指针 TreeNode* node
  2. 递归终止条件：当前节点为空节点
  3. 每层递归处理逻辑：
  • 处理当前节点（打印节点值、储存到数组中……）
  • 递归遍历左节点
  • 递归遍历右节点

迭代1​

操作节点顺序为中左右，当节点操作时，就是它弹出栈的时候。每次访问的时候的结点就是中节点，先做处理。

  1. 首先将根节点压入栈。
  2. while循环条件为栈不为空。
  3. 循环内：弹出当前栈顶节点为curr（操作这个节点）——压入右节点（不为空的情况下）——压入左节点（不为空的情况下）。

  • 94.中序遍历（左中右）​

递归​

  1. 递归传入参数为：当前节点的指针 TreeNode* node
  2. 递归终止条件：当前节点为空节点
  3. 每层递归处理逻辑：
  • 递归遍历左节点
  • 处理当前节点（打印节点值、储存到数组中……）
  • 递归遍历右节点

迭代1​

操作顺序为左中右，访问的结点为中，但是需要操作的结点为左。因此需要指针协助指向要操作的结点。

  1. 指针首先指向根节点。
  2. while循环条件：当前节点不为空或者栈内还有元素。
  3. 循环内：如果当前节点不为空，就将当前节点压入栈顶，并且令指针指向左节点；如果当前节点为空 —— 就先将栈顶弹出作为当前节点，并处理当前节点【当前处理的一定是由左边回退到更高一级的根节点】—— 然后将当前指针指向当前这个结点右节点进行处理。

  • 145.后序遍历（左右中）​

递归​

  1. 递归传入参数为：当前节点的指针 TreeNode* node
  2. 递归终止条件：当前节点为空节点
  3. 每层递归处理逻辑：
  • 递归遍历左节点
  • 递归遍历右节点
  • 处理当前节点（打印节点值、储存到数组中……）

迭代1​

后续遍历的顺序为左右中，发现可以从右边做一个访问节点及为处理节点的顺序，最后翻一下顺序就可以。

  1. 将当前节点压入栈。
  2. 循环条件为当前栈不为空。
  3. 循环内：弹出栈顶 —— 处理栈顶节点 —— 将当前节点的左节点压入栈（如果存在）——将当前节点的右节点压入栈（如果存在）。
  4. 将访问顺序反序输出即可。

  • 102.二叉树层序遍历​

借助队列完成对节点一层一层的遍历。

  1. 声明队列，类型为node类型。声明记录节点的容器。
  2. 如果根节点不为null，加入队列。
  3. 进入while循环，while条件：队列不为空。
  4. while：获取当前队列的长度（就是本层内节点的个数）—— 循环取出本层每一个节点—— 将值追加到本层的vector —— 如果当前节点->left不为null，则加入队列 —— 如果当前节点->right不为null，则加入队列 —— for循环之后将这一层的结点vector加入结果的vector。

一次while就是一层，while内循环的次数就是一层内节点的个数。

vector<vector<int>> levelOrder(TreeNode* root) {
        queue<TreeNode*> a;
        vector<vector<int>> result;
        if (root != nullptr)
            a.push(root);
        while (a.size() != 0) {
            int len = a.size();
            vector<int> temp;

                TreeNode* curr = a.front();
                temp.push_back(curr->val);
                a.pop();
                if (curr->left) 
                    a.push(curr->left);
                if (curr->right)
                    a.push(curr->right);
            }
            result.push_back(temp);
            // delete temp;
        }
        return result;
    }

3.199.二叉树的右视图

正常层序遍历树，在每一次while循环中，在for循环的最后一次将当前节点的值压入结果容器。

ps：需要注意的是右视图看到的不一定是哪棵树，也不一定是哪一个节点，因为不确定性，因此需要每次都遍历所有节点【所有节点都压入队列】，而不能认为有右子树就直接扔掉左子树（反例：左树比右树深）。

vector<int> rightSideView(TreeNode* root) {
        vector<int> result;
        queue<TreeNode*> a;
        if (root != nullptr) {
            a.push(root);
        }
        int len = a.size();
        while (len != 0) {
            vector<int> temp;
            for (int i = 0; i < len; i++) {
                TreeNode* curr = a.front();
                a.pop();
                temp.push_back(curr->val);
                if (curr->left) 
                    a.push(curr->left);
                if (curr->right)
                    a.push(curr->right);
            }
            result.push_back(temp.back());
            len = a.size();
        }
        return result;
    }

4.637.二叉树的层平均值

层次遍历。

在每一次while循环中，遍历本层每一个节点时将结点值求和。for循环完成，将和/本层节点个数，得出结果存到结果集。

PS：因为结果集是一个double集，存数据的时候需注意：

double = int/int => int相除还是int【去尾，. 后的不保留】，然后强转double。

double = double/int 或 int/double => 都变成double相除，得出结果也是double。 √（符合本题要求）

vector<double> averageOfLevels(TreeNode* root) {
        vector<double> result;
        queue<TreeNode*> a;
        if (root != nullptr) {
            a.push(root);
        }
        while(a.size() != 0) {
            double sum = 0;
            int len = a.size();
            for (int i = 0; i < len; i++) {
                TreeNode* curr = a.front();
                a.pop();
                sum += curr->val;
                if (curr->left) 
                    a.push(curr->left);
                if (curr->right)
                    a.push(curr->right);
            }
            double aver = sum / len;
            result.push_back(aver);
        }
        return result;
    }

5.515.层序遍历求最大值

层序遍历。

在每一次while循环中，遍历本层的每一个节点时不断更新最大值。每一次for循环结束就将最大值追加到结果集。

PS：默认的初始最大值为int表示范围的最小值：-2^31=0x80000000、

vector<int> largestValues(TreeNode* root) {
        vector<int> result;
        queue<TreeNode*> a;
        if (root != nullptr) 
            a.push(root);
        while (a.size() != 0) {
            int len = a.size();
            int maxx = 0x80000000;
            for (int i = 0; i < len; i++) {
                TreeNode* curr = a.front();
                a.pop();
                if (curr->val > maxx)
                    maxx = curr->val;
                if (curr->left)
                    a.push(curr->left);
                if (curr->right)
                    a.push(curr->right);
            }
            result.push_back(maxx);
        }
        return result;
    }

6.116.为每一个节点填充右侧节点的指针

层序遍历。

在每一次while循环中，每一层设置两个指针记录上一个节点和本节点：

先将当前节点的不为null的子节点加入队列，

判断当前节点是否为除第一个节点外的其他节点，如果是则让last的next赋值为curr节点，并且更新last节点为curr；

判断当前节点是最后一个节点，当前节点的next为null。

Node* connect(Node* root) {
        queue<Node*> a;
        if (root != NULL) {
            a.push(root);
        }
        while (a.size() != 0) {
            int len = a.size();
            Node* curr = a.front();
            Node* last = a.front();
            for (int i = 0; i < len; i++) {
                curr = a.front();
                a.pop();
                if (curr->left)
                    a.push(curr->left);
                if (curr->right)
                    a.push(curr->right);
                if (i > 0 && i < len) {
                    last->next = curr;
                    last = curr;
                } 
                if (i == len - 1) {
                    curr->next = NULL;
                } 
            }
        }
        return root;
    }

8.111.二叉树的最小深度

层序遍历。

比最大深度要多做一点工作。

while循环一次深度+1。但是每层操作节点的时候要判断是不是叶子结点，如果是则直接退出循环即可，层次遍历到叶子结点就是最小深度后面的不用再遍历。

叶子结点：左子节点和右子结点同时为null。

int minDepth(TreeNode* root) {
    queue<TreeNode*> a;
    int result = 0;
    bool flag = 0;
    if (root != 0) 
        a.push(root);
    while (a.size() != 0) {
        int len = a.size();
        result++;
        for (int i = 0; i < len; i++) {
            TreeNode* curr = a.front();
            a.pop();
            if (!curr->left && !curr->right) {
                flag = 1;
                break;
            }
            if (curr->left)
                a.push(curr->left);
            if (curr->right)
                a.push(curr->right);
        }
        if (flag == 1)
            break;
    }
    return result;
}

9.226.翻转二叉树

翻转二叉树，就是将每个节点的左右结点互换。

遍历时将其左右子结点翻转，遍历方式可以是层序遍历也可以是前中后序遍历。

这个时候对节点的操作就是交换左右子节点。

PS：中序遍历的时候，递归遍历顺序为左中右，在遍历中间结点的时候处理节点——交换左右子节点。这个时候左边已经处理完了，因此后面换过去之后应该不用处理右节点，应该还需处理左节点。

void preorder(TreeNode* node) {
        if (node == nullptr || (node->left == nullptr && node->right == nullptr) )
            return;
        else {
            TreeNode* temp = node->left;
            node->left = node->right;
            node->right = temp;
            // cout<<node->val<<" "<<node->left->val<<" "<<node->right->val<<endl;
            preorder(node->left);
            preorder(node->right);
        }

    }

TreeNode* invertTree(TreeNode* root) {
        // 前序遍历
        preorder(root);
        return root;
}

TreeNode* invertTree(TreeNode* root) {
         // 层序遍历
         queue<TreeNode*> a;
         if (root != nullptr)
               a.push(root);
         while (a.size() != 0) {
             int len = a.size();
             for (int i = 0; i < len; i++) {
                 TreeNode* curr = a.front();
                 a.pop();
                 TreeNode* temp = curr->left;
                 curr->left = curr->right;
                 curr->right = temp;
                 if (curr->left)
                     a.push(curr->left);
                 if (curr->right)
                     a.push(curr->right);
             }
         }
         return root;
    }

10.101.对称二叉树

乍一看有点像翻转二叉树，但不太一样，本题可以说是将原树进行反翻转之后判断是否和原树一样；也可以理解为根节点的左右子树是否互为翻转二叉树。

判断方式就是判断根节点的左右两棵子树【根节点分别为root->left，root->right】的对应位置是否节点完全一致。

两棵子树的对应位置分为内侧、外侧：

外侧对应位置：left->left —— right->right

内侧对应位置：left->right —— right->left

对应节点完全一致：

同时为空，或者值相同。

递归写法​

传参：当前两棵树节点指针left，right

返回值：bool 是否是对称二叉树，是true，否false

终止条件：当前根节点都为空，return true；当前的节点一个为空，另一个不为空，return false；当前节点都不为空，且值不相等，return false；

每层逻辑：当前节点都不为空，值相等时，递归判断外侧和内侧节点是否对应一致，返回节点为内侧结果&&外侧结果。

 bool compareLR(TreeNode* left, TreeNode* right) {
        if (left == nullptr && right == nullptr)
            return true;
        else if ((left == nullptr && right != nullptr) || (left != nullptr && right == nullptr))
            return false;
        else if (left->val != right->val)
            return false;
        bool outside = compareLR(left->left, right->right);
        bool inside = compareLR(left->right, right->left);
        return outside && inside;
            
    }

bool isSymmetric(TreeNode* root) {
        if (root != nullptr)
            return compareLR(root->left, root->right);
        return true;
    }

迭代写法​

利用栈或者队列，模仿递归或者层序遍历

将对应位置节点分别压入，弹出作比较。

12.110.平衡二叉树

判断左右子树高度差是否 >1, 是则return false，否则一直true。

递归：任意遍历顺序。

参数：当前处理节点

返回值：bool，true是平衡二叉树，false：不是。

结束条件：当前节点为空，返回true；当前节点左右子节点的高度差大于1，返回false；

每层逻辑：判断完是否为返回条件，如果不是，递归判断左右子节点是否符合，返回左右子树的判断结果的与。

int getHeight(TreeNode* node) {
        if (node == nullptr)
            return 0;

        int right = getHeight(node->right);
        return max(left, right) + 1;

13.257.二叉树的所有路径

题目的含义，就是记录，根节点到所有叶子结点的路径。路径使用字符串表示。

递归写法（前序遍历）​

传参：当前要访问的节点（必不为空），到目前为止路径字符串，完整路径字符串的集合引用。

返回值：空

终止条件：当前节点是叶子结点（左右子节点都为空），代表到了一条路径的终点，将这个路径记录到集合中。

每层逻辑：查看是否有左子节点，有则将左子节点、当前路径字符串+左子节点、和当前的字符串集合传入，递归遍历。同样查看右子节点并遍历。【保证不会传入空节点，传入的路径字符串已经包含当前传入节点】

调用递归函数时：需要先排除root为空的情况；以及需要传入root时将节点加入路径。

void preOrder(TreeNode* root,vector<string>& result,string curr) {

}

迭代写法（前序遍历）​

使用栈存储访问节点的同时，需要用同样的字符串栈，同步存储以当前节点为结尾的路径字符串。

每次while循环：从栈中取出当前节点的同时，同步取出当前节点对应的路径字符串。判断当前是叶子节点时，就将路径加入结果集合。如果不是则根据当前左子节点、右子结点是否存在，将其压入栈，同步将字符串补上当前左子节点/右子结点的值压入路径栈。

vector<string> binaryTreePaths(TreeNode* root) {
    // 迭代
        stack<TreeNode*> s;
        stack<string> str;
        vector<string> res;
        if (root ==  nullptr) return res;
        s.push(root);
        str.push(to_string(root->val));
        while (!s.empty()) {
            TreeNode* curr = s.top();
            s.pop();
            string curStr = str.top();
            str.pop();
            if (curr->left == nullptr && curr->right == nullptr) {
                res.push_back(curStr);
                continue;
            }
            if (curr->right) {
                s.push(curr->right);
                str.push(curStr + "->" + to_string(curr->right->val));
            }
            if (curr->left) {
                s.push(curr->left);
                str.push(curStr + "->" + to_string(curr->left->val));
            }
        }
        return res;
    }

14.404. 左叶子之和

递归写法​

传参：当前要访问的节点

返回值：int，当前节点为根节点的左叶子和。

终止条件：当前节点是空结点，返回0。

每层逻辑：

当前节点包含左叶子节点：返回左叶子节点值+右节点递归左叶子节点和 = 当前节点为根节点时的左叶子之和。return和。

为什么这样写会有空指针问题？

/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    void buildDG (TreeNode* root,vector<int> inorder, vector<int> postorder) {
        if (postorder.size() == 0) {
            root = nullptr;
            return ;
        }
        int len = postorder.size();
        int val = postorder[len - 1];
        int target = 0;
        vector<int> inorder_l;
        vector<int> postorder_l;
        for (int i = 0; i < len ; i++) {
            if (inorder[i] == val) {
                target = i;
                break;
            }
            inorder_l.push_back(inorder[i]);
            postorder_l.push_back(postorder[i]);
        }
        root = new TreeNode(val);
        inorder.erase(inorder.begin(),inorder.begin() + target + 1);
        postorder.erase(postorder.begin(),postorder.begin() + target);
        postorder.pop_back();
        buildDG(root->left, inorder_l,postorder_l);
        buildDG(root->right, inorder,postorder);
        return ;
    }
    TreeNode* buildTree(vector<int>& inorder, vector<int>& postorder) {
        TreeNode* root;
        buildDG(root,inorder,postorder);
        return root;
    }
};
